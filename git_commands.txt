1. git clone [url] - клонирует удаленный репозиторий на компьютер.
2. git status - показывает состояние файлов.
3. git add [файл] - добавляет файл в очередь на коммит.
4. git commit -m "текст" - сохраняет изменения с комментарием.
5. git push - отправляет изменения на GitHub.
6. git pull - загружает изменения с сервера.
7. git log - показывает историю коммитов.
8. git branch - показывает список веток.
9. git checkout -b [имя] - создает новую ветку.
10. git init - создает новый локальный репозиторий.
